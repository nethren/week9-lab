name: pipeline

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.13"
      - run: pip install ruff
      - run: ruff check .

  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.13"
      - run: pip install -r requirements.txt
      - run: python -m unittest discover -v

  audit:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.13"
      - run: pip install pip-audit
      - run: pip-audit -r requirements.txt

  package:
    needs: [ lint, test, audit ]
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.13"
      - name: Build the deployable zip
        run: |
          pip install -r requirements.txt -t build/
          cp app.py duckfine.py lambda_function.py build/
          cd build && zip -qr ../quackloan.zip .
      - name: Generate an SBOM
        uses: anchore/sbom-action@v0
        with:
          path: build/
          format: spdx-json
          artifact-name: sbom.spdx.json
      - uses: actions/upload-artifact@v4
        with:
          name: quackloan-deployment-package
          path: quackloan.zip

  smoke:
    needs: package
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.13"
      - uses: actions/download-artifact@v4
        with:
          name: quackloan-deployment-package
      - name: Run the artifact (not the source) and smoke-test it
        run: |
          unzip -q quackloan.zip -d staged
          cd staged
          python -m flask --app app run > ../app.log 2>&1 &
          cd ..
          bash smoke.sh http://localhost:5000 || { cat app.log; exit 1; }

  e2e:
    needs: package
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.13"
      - uses: actions/download-artifact@v4
        with:
          name: quackloan-deployment-package
      - name: Install Playwright and its browser
        run: |
          pip install playwright
          python -m playwright install --with-deps chromium
      - name: Run the artifact and test it through a browser
        run: |
          unzip -q quackloan.zip -d staged
          cd staged
          python -m flask --app app run > ../app.log 2>&1 &
          cd ..
          for i in $(seq 1 20); do
            curl -fs http://localhost:5000/health > /dev/null && break
            sleep 1
          done
          python -m unittest discover -s e2e -v \
            || { cat app.log; exit 1; }

  staging:
    needs: [ smoke, e2e ]
    if: github.event_name == 'push'
    runs-on: ubuntu-latest
    environment: staging
    env:
      AWS_ACCESS_KEY_ID: ${{ secrets.AWS_ACCESS_KEY_ID }}
      AWS_SECRET_ACCESS_KEY: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
      AWS_SESSION_TOKEN: ${{ secrets.AWS_SESSION_TOKEN }}
      AWS_DEFAULT_REGION: us-east-1
    steps:
      - uses: actions/checkout@v4
      - uses: actions/download-artifact@v4
        with:
          name: quackloan-deployment-package
      - name: Deploy the same zip
        run: |
          aws lambda update-function-code \
            --function-name "${{ vars.FUNCTION_NAME }}" \
            --zip-file fileb://quackloan.zip \
            --query CodeSha256 --output text
          aws lambda wait function-updated \
            --function-name "${{ vars.FUNCTION_NAME }}"
      - name: Smoke-test what was deployed
        run: bash smoke.sh "${{ vars.FUNCTION_URL }}"

  production:
    needs: staging
    runs-on: ubuntu-latest
    environment: production
    env:
      AWS_ACCESS_KEY_ID: ${{ secrets.AWS_ACCESS_KEY_ID }}
      AWS_SECRET_ACCESS_KEY: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
      AWS_SESSION_TOKEN: ${{ secrets.AWS_SESSION_TOKEN }}
      AWS_DEFAULT_REGION: us-east-1
    steps:
      - uses: actions/checkout@v4
      - uses: actions/download-artifact@v4
        with:
          name: quackloan-deployment-package
      - name: Deploy the same zip
        run: |
          aws lambda update-function-code \
            --function-name "${{ vars.FUNCTION_NAME }}" \
            --zip-file fileb://quackloan.zip \
            --query CodeSha256 --output text
          aws lambda wait function-updated \
            --function-name "${{ vars.FUNCTION_NAME }}"
      - name: Smoke-test what was deployed
        run: bash smoke.sh "${{ vars.FUNCTION_URL }}"
